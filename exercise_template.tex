% ==== EXERCISE / SOLUTION STRUCTURE ====
\newcounter{exercise}

% Problem heading (you can optionally put the problem text after it)
\newenvironment{exercise}[1][]{
	\refstepcounter{exercise}%
	\par\noindent\textbf{Problem \theexercise. #1}\par\smallskip
}{\par\medskip}

% You said you will only write the solutions
\newenvironment{solution}{
	\par\noindent\textbf{Solution. }\ignorespaces
}{\par\medskip}

% ==== MACROS FOR IMAGES & "GRAPHS" (PLOTS) ====
% General figure macro
% Usage: \insfig[width]{filename}{Caption text}{label-name}
% Example: \insfig[0.6\textwidth]{myplot}{Sample plot}{sample-plot}
\newcommand{\insfig}[4][0.8\textwidth]{%
	\begin{figure}[h]
		\centering
		\includegraphics[width=#1]{#2}
		\caption{#3}
		\label{fig:#4}
	\end{figure}
}

% Alias specifically for graphs/plots (same as image, different default width)
% Usage: \insgraph{filename}{Caption text}{label-name}
\newcommand{\insgraph}[3]{%
	\insfig[0.7\textwidth]{#1}{#2}{#3}%
}
% SVG figure macro
% Usage: \inssvg[width]{filename-without-ext}{Caption text}{label-name}
% Example: \inssvg[0.7\textwidth]{diagram}{An SVG diagram}{svg-diagram}
\newcommand{\inssvg}[4][0.8\textwidth]{%
	\begin{figure}[h]
		\centering
		\includesvg[width=#1,inkscapelatex=false]{#2}
		\caption{#3}
		\label{fig:#4}
	\end{figure}
}
